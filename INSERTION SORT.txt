
//**************************INSERTION SORT************************
#include<iostream>
using namespace std;
int main(){
    int arr[5]={10,54,75,62,5};

    for(int i=1;i<5;i++){

        int j=i-1;
        int temp=arr[i];

        for(;j>=0;j--){

            if(arr[j]>temp){
                arr[j+1]=arr[j];
            }
            else{
                break;
            }

        }

        arr[j+1]=temp;
    }

    for(int i=0;i<5;i++){

        cout<<arr[i]<<" ";
    }
}
