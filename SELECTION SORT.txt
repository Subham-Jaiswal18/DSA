
//****************************SELECTION SORT********************
#include<iostream>
using namespace std;
int main(){


    int arr[5]={10,3,5,7,8};

    for(int i=0;i<4;i++){


        int minindex=i;

        for(int j=i;j<5;j++){

            if(arr[minindex]>arr[j]){
                minindex=j;
            }

        }
        swap(arr[minindex],arr[i]);
    }

    for(int i=0;i<5;i++){
        cout<<arr[i]<<" ";
    }

}
