void saydigit(int n,string arr[]){
    if(n==0){
        return;
    }
    int digit=n%10;
    n/=10;
    saydigit(n,arr);
    cout<<arr[digit]<<" ";                       //BAAD ME ISLIYE KYUKI JB BASE CONDITION REACH HOGI FIR USKE BAAD NUMBER LAST SE PRINT                                                                   /                                                 /HONGE AGAR NHI lagate to reverse me print hota hai
  }

};
