class Solution {
public:
    int minimumDeletions(vector<int>& nums) {
        int n=nums.size();
        int minIndex=0;
        int maxIndex=0;
        for(int i=0;i<n;i++){
            if(nums[i]>nums[maxIndex]){
                maxIndex=i;
            }
            if(nums[i]<nums[minIndex]){
                minIndex=i;
            }
        }
        int left=min(maxIndex,minIndex);
        int right=max(maxIndex,minIndex);
        int leftSide=right+1;
        int rightSide=n-left;
        int individual=(left+1)+(n-right);
        return min({leftSide,rightSide,individual});

    }
};
