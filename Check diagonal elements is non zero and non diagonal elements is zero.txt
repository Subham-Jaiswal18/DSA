class Solution {
public:
    bool checkXMatrix(vector<vector<int>>& grid) {
        int size=grid.size();
        
        for(int i=0;i<size;i++){
            for(int j=0;j<size;j++){
                bool isdiagonal= (i==j) || (i+j==size-1);
                if(isdiagonal ? (grid[i][j]==0) : (grid[i][j]!=0)){
                    return false;
                }
                }
        }
        return true;
    }
};
