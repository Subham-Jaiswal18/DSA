class Solution {
public:
    int maximumLengthSubstring(string s) {
        if(s.length()<2){
            return s.length();
        }
        int left=0;
        int count[26]={0};
        int max_length=0;
        for(int i=0;i<s.length();i++){
            count[s[i]-'a']++;
            while(count[s[i]-'a']>2){
                count[s[left]-'a']--;
                left++;
            }
            max_length=max(max_length,i-left+1);

        }
        return max_length;
    }
};
