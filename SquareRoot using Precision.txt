#include<iostream>
using namespace std;

long long int sqrt(int n){
    int start=0;
    int end=n;
    long long int mid=start+(end-start)/2;
    long long ans=-1;
    while(start<=end){
        if(mid*mid<=n){
            ans=mid;
            start=mid+1;
        }
        else{
            end=mid-1;
        }
        mid=start+(end-start)/2;
    }
    return ans;
}
double precise(int n,int temp,int p){
    double fact=1;
    double ans=temp;
    for(int i=0;i<p;i++){
        fact=fact/10;
    for(double j=ans;j*j<n;j+=fact){
       ans=j; 
    }
}
    return ans;
}


int main(){
 int n;
    int p;
   cout<<"Enter the number :";
   cin>>n;
   cout<<"Enter the precision :";
   cin>>p;
   int temp=sqrt(n);
   cout<<precise(n,temp,p);
}
