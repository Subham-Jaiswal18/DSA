class Solution {
public:
    int sumOfDigit(int n){
        if(n<=9){
            return n;
        }
        int sum=0;
        while(n!=0){
            int digit=n%10;
            sum+=digit;
            n/=10;
        }
    return sumOfDigit(sum);
    }
    int addDigits(int num) {
        return sumOfDigit(num);
    }
};
