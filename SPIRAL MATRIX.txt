class Solution {
public:
    vector<int> spiralOrder(vector<vector<int>>& matrix) {
       vector<int>ans;
       int row=matrix.size();
       int col=matrix[0].size();
       int startRow=0;
       int endRow=row-1;
       int startCol=0;
       int endCol=col-1;
       int value=row*col;
       int count=0;
       while(value>count){

        // START ROW

        for(int i=startCol;i<=endCol && value>count;i++){
            ans.push_back(matrix[startRow][i]);
            count++;
        }
        startRow++;

        //END COLUMN

        for(int j=startRow;j<=endRow && value>count;j++){
            ans.push_back(matrix[j][endCol]);
            count++;
        }
        endCol--;

        //END ROW

        for(int k=endCol;k>=startCol && value>count;k--){
            ans.push_back(matrix[endRow][k]);
            count++;
        }
        endRow--;

        //START COLUMN

        for(int l=endRow;l>=startRow && value>count;l--){
            ans.push_back(matrix[l][startCol]);
            count++;
        }
        startCol++;
       }
       return ans;
    }
};
