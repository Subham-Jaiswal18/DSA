
//****************************Bubble sort*******************

#include<iostream>
using namespace std;
int main(){

    int arr[5]={10,1,8,9,6};

    bool isswap=false;

    for(int i=0;i<4;i++){

        for(int j=0;j<5-i-1;j++){

            if(arr[j]>arr[j+1]){
                swap(arr[j],arr[j+1]);
            }

            isswap=true;
    }

    if(isswap=false){
        break;
    }

}

for(int i=0;i<5;i++){
    cout<<arr[i]<<" ";
}

}
