class Solution {
public:
    bool searchMatrix(vector<vector<int>>& matrix, int target) {
        int row=matrix.size()-1;
        int col=matrix[0].size()-1;
        int startrow=0;
        int end=row*col-1;
      
        while(row>=startrow && 0<=col){
            int element=matrix[startrow][col];
            if(element==target){
                return true;
            }
            else if(element>target){
                col--;
            }
            else{
                startrow++;
            }
           
        }
        return false;
    }
};
