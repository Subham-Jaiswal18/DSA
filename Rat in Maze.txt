class Solution {

    private:

        bool isSafe(int x,int y,int size,vector<vector<int>>visited,vector<vector<int>>& maze){
           
            if((x>=0 && x<=size-1) && (y>=0 && y<=size-1) && visited[x][y]==0 && maze[x][y]==1){
                return true;
            }
            else{
                return false;
            }
        }

       void solve(vector<vector<int>>& maze,int size,int x,int y,vector<vector<int>>visited,vector<string>&ans,string op){

           if(x==size-1 && y==size-1){
               ans.push_back(op);
               return;
           }
           visited[x][y]=1;

               //UP
               int newx=x-1;
               int newy=y;
               if(isSafe(newx,newy,size,visited,maze)){
               op.push_back('U');
               solve(maze,size,newx,newy,visited,ans,op);
               op.pop_back();
               }


               //DOWN


               newx=x+1;
              newy=y;
               if(isSafe(newx,newy,size,visited,maze)){
               op.push_back('D');
               solve(maze,size,newx,newy,visited,ans,op);
               op.pop_back();
               }


               //LEFT

               newx=x;
                newy=y-1;
               if(isSafe(newx,newy,size,visited,maze)){
               op.push_back('L');
               solve(maze,size,newx,newy,visited,ans,op);
               op.pop_back();
               }
               //RIGHT

             newx=x;
                newy=y+1;
               if(isSafe(newx,newy,size,visited,maze)){
               op.push_back('R');
               solve(maze,size,newx,newy,visited,ans,op);
               op.pop_back();
               }



           visited[x][y]=0;
       } 



  public:

    vector<string> ratInMaze(vector<vector<int>>& maze) {
        // code here
        int size=maze.size();
        vector<string>ans;
        if(maze[0][0]==0){
            return ans;
        }
        int x=0;
        int y=0;
        vector<vector<int>>visited=maze;
        for(int i=0;i<size;i++){
            for(int j=0;j<size;j++){
                visited[i][j]=0;
            }
        }
        string op="";
        solve(maze,size,x,y,visited,ans,op);
        sort(ans.begin(),ans.end());
        return ans;
    }
};
