class Solution {
public:
    char nextGreatestLetter(vector<char>& letters, char target) {
        char ans=letters[0];
        int start=0;
        int end=letters.size()-1;
        int mid=start+(end-start)/2;
        while(start<=end){
            if(letters[mid]>target){
                ans=letters[mid];
                end=mid-1;
            }
            else{
                start=mid+1;
            }
            mid=start+(end-start)/2;
        }
        return ans;
    }
};
