long long int sqrt(int n){
    int start=0;
    int end=n;
    long long int mid=start+(end-start)/2;
    long long ans=-1;
    while(start<=end){
        if(mid*mid<=n){
            ans=mid;
            start=mid+1;
        }
        else{
            end=mid-1;
        }
        mid=start+(end-start)/2;
    }
    return ans;
}
class Solution {
public:
    int mySqrt(int x) {
        return sqrt(x);
    }
};
