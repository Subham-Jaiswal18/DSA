class Solution {
public:
    bool validMountainArray(vector<int>& arr) {
       int i=0;
       int size=arr.size()-1;
       while(i<size && arr[i]<arr[i+1]){
        i++;
       }
       if(i==0 || i==size){
        return false;
       }

       while(i<size && arr[i+1]<arr[i]){
        i++;
       }
       return i==size;

    }
};
