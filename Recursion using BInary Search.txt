#include<iostream>
using namespace std;
void print(int arr[],int s,int e){
    for(int i=s;i<=e;i++){
        cout<<arr[i]<<" ";
    }
    cout<<endl;
}
bool binarySearch(int arr[],int s,int e,int k){
    print(arr,s,e);
   if(s>e){
       return false;
   }
   int mid=s+(e-s)/2;
   if(arr[mid]==k){
       return true;
   }
   if(arr[mid]>k){
       return binarySearch(arr,0,mid-1,k);
   }
   else{
       return binarySearch(arr,mid+1,e,k);
   }
   return false;
    
}
int main(){
    int arr[5]={45,58,98,99,455};
    int s=0;
    int e=4;
    int k=58;
    cout<<binarySearch(arr,s,e,k);
    
}
