class Solution {
public:
    int maxProduct(vector<int>& nums) {
        int product=nums[0];
        int suffix=0;
        int prefix=0;
        for(int i=0;i<nums.size();i++){
            prefix=(prefix==0 ? 1 : prefix )*nums[i];
            suffix=(suffix==0 ? 1 : suffix )*nums[nums.size()-i-1];
             product=max({prefix,suffix,product});

        }
       
        return product;
    }
};
