bool ispossible(vector<int> &boards, int k,long long mid){
    long long boardsum=0;
    int painter=1;
    for(int i=0;i<boards.size();i++){
        if(boards[i]+boardsum<=mid){
            boardsum+=boards[i];
        }
        else{
            painter++;
            if(painter>k||boards[i]>mid){
                return false;
            }
            boardsum=boards[i];
        }
    }
    return true;

}

int findLargestMinDistance(vector<int> &boards, int k)
{
    //    Write your code here.
   
    long long start=0;
    long long sum=0;
    for(int i=0;i<boards.size();i++){
        sum+=boards[i];
    }
    long long end=sum;
    long long mid=start+(end-start)/2;
    long long ans=-1;
    while(start<=end){
        if(ispossible(boards,k,mid)){
            ans=mid;
            end=mid-1;
        }
        else{
            start=mid+1;

        }
        mid=start+(end-start)/2;
    }
    return ans;
}
