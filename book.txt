bool ispossible(vector<int> time,int n,long long mid){
    int chaptercount=1;
    long long pagesum=0;
    for(int i=0;i<time.size();i++){
        if(pagesum+time[i]<=mid){
            pagesum+=time[i];
        }
        else{
            chaptercount++;
            if(chaptercount>n||time[i]>mid){
                return false;
            }
            pagesum=time[i];
        }
    }
    return true;
}

long long ayushGivesNinjatest(int n, int m, vector<int> time) 
{	
	// Write your code here.
    long long start=0;
    long long sum=0;
    for(int i=0;i<time.size();i++){
        sum+=time[i];
    }
    long long end=sum;
    long long ans=-1;
    long long mid=start+(end-start)/2;
    while(start<=end){
        if(ispossible(time,n,mid)){
            ans=mid;
            end=mid-1;
        }
        else{
            start=mid+1;
        }
        mid=start+(end-start)/2;
    }
    return ans;
}
