class Solution {
public:

    double power(double x,long long n){
        if(n==0){
            return 1;
        }
        double ans=power(x,n/2);
        if(n&1){
            return ans*ans*x;
        }
        else{
            return ans*ans;
        }
    }
    double myPow(double x, int n) {
        long long N=n;
        if(N<0){
            N=-N;
            x=1.0/x;
        }
        return power(x,n);
    }
};
