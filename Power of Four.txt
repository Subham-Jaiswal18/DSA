class Solution {
public:
    bool isPowerOfFour(int n) {
        // 1. Must be positive
        // 2. Must be a power of two: (n & (n - 1)) == 0
        // 3. The single '1' bit must be at an even bit position: (n & 0x55555555) != 0
        return n > 0 && (n & (n - 1)) == 0 && (n & 0x55555555);
    }
};
