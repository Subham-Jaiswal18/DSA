class Solution {
public:


    bool ispossible(vector<int>& piles, int h,int mid,int maxi){
        if(mid>maxi){
            return false;
        }
        else{
        long long count=0;
        for(int i=0;i<piles.size();i++){
            int div=ceil(double(piles[i])/mid);
            count+=div;
        }
        if(count>h){
            return false;
        }
    }
        return true;
    }
    int minEatingSpeed(vector<int>& piles, int h) {
        int start=1;
        int maxi=*max_element(piles.begin(),piles.end());
        int end=maxi;
        int mid=start+(end-start)/2;
        int ans=-1;
        while(start<=end){
            if(ispossible(piles,h,mid,maxi)){
                ans=mid;
                end=mid-1;
            }
            else{
                start=mid+1;
            }
            mid=start+(end-start)/2;
        }
        return ans;
    }
};
