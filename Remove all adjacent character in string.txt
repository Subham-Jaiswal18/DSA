class Solution {
public:
    string removeDuplicates(string s) {

        string res = "";

        for (char c : s) {

            if (!res.empty() && res.back() == c) {

                res.pop_back(); // O(1) time
            }  

	    else {

                res.push_back(c); // O(1) time

            }

        }
        return res;
    }
};
